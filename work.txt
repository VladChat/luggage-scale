# Проверяем, что на нужной ветке
git checkout main

# Подтягиваем последние изменения из GitHub (чтобы точно совпадало)
git pull origin main

# Создаём тег с именем (например, snapshot-2025-10-01)
git tag -a snapshot-2025-10-02 -m "posts works October 2 2025"

# Отправляем тег на GitHub
git push origin snapshot-2025-10-02
